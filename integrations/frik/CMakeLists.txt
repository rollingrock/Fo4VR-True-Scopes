cmake_minimum_required(VERSION 4.2)

if(NOT DEFINED VCPKG_ROOT AND DEFINED ENV{VCPKG_ROOT})
  set(VCPKG_ROOT "$ENV{VCPKG_ROOT}")
endif()
if(DEFINED VCPKG_ROOT)
  set(CMAKE_TOOLCHAIN_FILE "${VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg toolchain")
endif()
set(VCPKG_TARGET_TRIPLET "x64-windows-static" CACHE STRING "vcpkg target triplet")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")

project(FO4VRScopeCalibration VERSION 0.7.0 LANGUAGES CXX RC)

set(F4VR_COMMON_FRAMEWORK_PATH "" CACHE PATH "F4VR-CommonFramework source")
if(NOT TARGET F4VRCommonFramework)
  if(NOT EXISTS "${F4VR_COMMON_FRAMEWORK_PATH}/src/ModBase.h")
    message(FATAL_ERROR "Set F4VR_COMMON_FRAMEWORK_PATH to the compatible framework checkout")
  endif()
  add_subdirectory("${F4VR_COMMON_FRAMEWORK_PATH}" framework EXCLUDE_FROM_ALL)
endif()
find_package(nlohmann_json CONFIG REQUIRED)

set(ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
configure_file(cmake/Version.h.in include/Version.h @ONLY)
configure_file(cmake/version.rc.in version.rc @ONLY)
file(TO_CMAKE_PATH "${ROOT_DIR}/data/config/FO4VRScopeCalibration.ini" CONFIG_INI)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" CONTENT "#include <windows.h>\n101 RCDATA \"${CONFIG_INI}\"\n")
add_library(FO4VRScopeCalibration SHARED src/FO4VRScopeCalibration.cpp src/Config.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc" "${CMAKE_CURRENT_BINARY_DIR}/resources.rc")
target_compile_features(FO4VRScopeCalibration PRIVATE cxx_std_23)
target_compile_definitions(FO4VRScopeCalibration PRIVATE _UNICODE _CRT_SECURE_NO_WARNINGS F4VR_WITH_OPENVR=1)
target_include_directories(FO4VRScopeCalibration PRIVATE src ../../src ../../src/TrueScopes "${CMAKE_CURRENT_BINARY_DIR}/include")
target_precompile_headers(FO4VRScopeCalibration PRIVATE src/PCH.h)
target_link_libraries(FO4VRScopeCalibration PRIVATE F4VRCommonFramework nlohmann_json::nlohmann_json)

install(TARGETS FO4VRScopeCalibration RUNTIME DESTINATION F4SE/Plugins)
install(DIRECTORY data/mod/ DESTINATION .)
install(FILES ../../LICENSE ASSET_CREDITS.md DESTINATION .)
